    let rowCount = 1;

    function addRow() {
        const container = document.getElementById('employesContainer');
        const firstRow = container.querySelector('.employe-row');
        const newRow = firstRow.cloneNode(true);

        newRow.dataset.index = rowCount;

        const selects = newRow.querySelectorAll('select');
        selects.forEach(s => s.selectedIndex = 0);

        const salaireGroup = newRow.querySelector('.salaire-group');
        salaireGroup.style.display = 'none';
        const salaireInput = salaireGroup.querySelector('.salaire-input');
        salaireInput.value = '';
        salaireInput.removeAttribute('name');

        const removeBtn = newRow.querySelector('.btn-remove-row');
        removeBtn.onclick = function() { removeRow(this); };

        container.appendChild(newRow);
        rowCount++;
    }

    function removeRow(btn) {
        const container = document.getElementById('employesContainer');
        const rows = container.querySelectorAll('.employe-row');
        if (rows.length <= 1) {
            alert('Vous devez garder au moins un employé.');
            return;
        }
        const row = btn.closest('.employe-row');
        row.remove();
    }

    document.getElementById('employesContainer').addEventListener('change', function(e) {
        if (e.target.tagName === 'SELECT' && e.target.name === 'idUtilisateur') {
            updateSalaryVisibility(e.target);
        }
    });

    function updateSalaryVisibility(select) {
        const row = select.closest('.employe-row');
        const salaireGroup = row.querySelector('.salaire-group');
        const salaireInput = salaireGroup.querySelector('.salaire-input');
        const selected = select.options[select.selectedIndex];
        const role = selected ? selected.getAttribute('data-role') : '';

        if (role === 'REMPLACANT') {
            salaireGroup.style.display = 'block';
            salaireInput.name = 'salaireJournalierRemplacant';
            salaireInput.required = true;
        } else {
            salaireGroup.style.display = 'none';
            salaireInput.removeAttribute('name');
            salaireInput.required = false;
            salaireInput.value = '';
        }
    }
</script>
